treeskel.winter_gt

Automatic main-branch ground truth from winter depth images.

In winter the trees are bare, so the main branches can be seen directly. This module extracts them from the winter depth image in five steps:

  1. Foreground (foreground_mask()). Above the ground line, rows are mostly "no reading" because the rows behind are out of range, so any valid pixel is foreground. In rows that are mostly ground, a pixel is foreground if it is clearly nearer (brighter) than that row's median, which is the ground.
  2. Tree selection (select_tree()). Choose the connected region that is large, reaches low in the image and is centred at its base: the tree being photographed rather than a neighbour or a post.
  3. Skeleton and gap linking (link_gaps()). Thin the tree to one-pixel centre lines. Depth often misses sections of branch, so a branch end that points upwards is joined to a nearby skeleton piece lying ahead of it, within a distance and angle limit.
  4. Main branches (find_main_branches()). The trunk root is the skeleton point at the bottom centre of the tree. The trees are trained as a 2D open V (Tatura trellis), so one leader is chosen on each side of the root: the tip that climbs highest, minus a penalty for wandering sideways. Candidates whose upper part is wider than a leader can be (trellis posts) are skipped. The fork is where the two leader paths separate.
  5. Quality check. The tree is flagged, so it can be left out rather than given a wrong label, if a leader rises less than a set fraction of the root's height (usually a depth gap), or if no non-post leader exists on a side. (The trunk's width is recorded but not checked: near the ground, ground pixels merge into the trunk and make every trunk look post-wide.)

render_main_branches() turns the result into a mask, with the width at each point taken from the tree's thickness there.

Run python -m treeskel.winter_gt DATA_DIR --out outputs/winter_gt to process a whole dataset folder (add --figure overview.png for a visual check).

182def foreground_mask(depth: np.ndarray, valid: np.ndarray, params: GTParams = GTParams()) -> np.ndarray:
183    """Find pixels that stand out from the ground or the empty background.
184
185    Args:
186        depth: ``(H, W)`` 8-bit depth (brighter = nearer).
187        valid: ``(H, W)`` bool, True where depth has a reading.
188        params: Settings.
189
190    Returns:
191        ``(H, W)`` bool foreground mask.
192    """
193    d = depth.astype(float)
194    fg = np.zeros(valid.shape, bool)
195    for y in range(d.shape[0]):
196        v = valid[y]
197        if v.mean() < params.ground_row_fraction:
198            fg[y] = v  # above the ground line: anything in range is foreground
199        else:
200            fg[y] = v & (d[y] > np.median(d[y][v]) + params.ground_margin)
201    return fg

204def select_tree(fg: np.ndarray, params: GTParams = GTParams()) -> np.ndarray:
205    """Choose the region most likely to be the photographed tree.
206
207    Each region is scored by ``area × (lowest row / height) × centring²``,
208    where centring is 1 when the base of the region (its lowest 10%) is at the
209    image centre and 0 at an edge.
210
211    Args:
212        fg: Foreground mask from :func:`foreground_mask`.
213        params: Settings (``min_tree_area``).
214
215    Returns:
216        Bool mask of the chosen region (all False if none is large enough).
217    """
218    lab, n = ndi.label(fg, structure=_K8)
219    h, w = fg.shape
220    best, best_score = 0, -1.0
221    for i in range(1, n + 1):
222        ys, xs = np.nonzero(lab == i)
223        if len(ys) < params.min_tree_area:
224            continue
225        base_x = np.median(xs[ys >= np.percentile(ys, 90)])
226        centring = 1 - abs(base_x - w / 2) / (w / 2)
227        score = len(ys) * (ys.max() / h) * centring**2
228        if score > best_score:
229            best, best_score = i, score
230    return lab == best if best else np.zeros_like(fg)

233def _graph(skel: np.ndarray) -> tuple[np.ndarray, np.ndarray, csr_matrix]:
234    """Pixel graph of a skeleton: node coordinates and 8-neighbour edge lengths."""
235    ys, xs = np.nonzero(skel)
236    idx = -np.ones(skel.shape, int)
237    idx[ys, xs] = np.arange(len(ys))
238    rows, cols, wts = [], [], []
239    for dy, dx in ((0, 1), (1, 0), (1, 1), (1, -1)):
240        y2, x2 = ys + dy, xs + dx
241        ok = (y2 < skel.shape[0]) & (x2 >= 0) & (x2 < skel.shape[1])
242        j = np.full(len(ys), -1)
243        j[ok] = idx[y2[ok], x2[ok]]
244        m = j >= 0
245        rows.append(np.nonzero(m)[0])
246        cols.append(j[m])
247        wts.append(np.full(int(m.sum()), np.hypot(dy, dx)))
248    n = len(ys)
249    a = coo_matrix((np.concatenate(wts), (np.concatenate(rows), np.concatenate(cols))), shape=(n, n))
250    return ys, xs, (a + a.T).tocsr()

253def _end_directions(skel: np.ndarray, back: float = 12.0) -> list[tuple[int, int, np.ndarray]]:
254    """Skeleton end points with the unit direction the branch is heading."""
255    ys, xs, a = _graph(skel)
256    out = []
257    for e in np.nonzero(np.diff(a.indptr) == 1)[0]:
258        dist = dijkstra(a, indices=e, limit=back)
259        f = int(np.argmax(np.where(np.isinf(dist), -1, dist)))
260        d = np.array([ys[e] - ys[f], xs[e] - xs[f]], float)
261        n = np.hypot(*d)
262        if n > 0:
263            out.append((int(ys[e]), int(xs[e]), d / n))
264    return out

267def link_gaps(main: np.ndarray, others: np.ndarray, params: GTParams = GTParams(),
268              iters: int = 15) -> np.ndarray:
269    """Join skeleton pieces that continue a branch across a depth gap.
270
271    A branch end heading upwards is joined by a straight line to the nearest
272    pixel of another piece within ``link_gap`` and within ``link_max_angle``
273    of the end's direction; that whole piece is then added. Repeats until
274    nothing more joins.
275
276    Args:
277        main: Skeleton of the tree.
278        others: Skeletons of other foreground pieces.
279        params: Settings.
280        iters: Maximum number of passes.
281
282    Returns:
283        The extended skeleton.
284    """
285    lab, _ = ndi.label(others, structure=_K8)
286    oy, ox = np.nonzero(others)
287    skel = main.copy()
288    cos_min = np.cos(np.radians(params.link_max_angle))
289    for _ in range(iters):
290        added = False
291        for y, x, d in _end_directions(skel):
292            if -d[0] < params.link_min_up:  # row index decreases going up
293                continue
294            dy, dx = oy - y, ox - x
295            dist = np.hypot(dy, dx)
296            ok = (dist > 0) & (dist <= params.link_gap) & ~skel[oy, ox]
297            ok &= (dy * d[0] + dx * d[1]) >= cos_min * dist
298            if not ok.any():
299                continue
300            k = np.nonzero(ok)[0][np.argmin(dist[ok])]
301            rr, cc = draw_line(y, x, int(oy[k]), int(ox[k]))
302            skel[rr, cc] = True
303            skel |= lab == lab[oy[k], ox[k]]
304            added = True
305        if not added:
306            break
307    return skel

310def path_shape(path: np.ndarray, radius: np.ndarray, top_fraction: float = 0.6) -> dict:
311    """Width, straightness and tilt of the upper part of a path.
312
313    The upper part is the points within the top ``top_fraction`` of the
314    path's rise. A straight line is fitted to it by total least squares (the
315    principal axis from an SVD of the centred points), which, unlike ordinary
316    regression, treats rows and columns alike and so also works for
317    near-vertical lines.
318
319    Args:
320        path: ``(N, 2)`` ``(row, col)`` points, starting at the bottom.
321        radius: Half-width image (distance from each pixel to the edge of the
322            tree region).
323        top_fraction: Fraction of the rise, from the top, to measure.
324
325    Returns:
326        ``{"radius": median half-width (px), "straightness": RMS distance from
327        the fitted line (px), "tilt": angle from vertical (degrees)}``, or an
328        empty dict if the path has fewer than 3 points.
329    """
330    p = np.asarray(path)
331    if len(p) < 3:
332        return {}
333    rise = p[0, 0] - p[:, 0].min()
334    q = p[p[:, 0] <= p[0, 0] - (1 - top_fraction) * rise]
335    if len(q) < 3:
336        q = p
337    c = q - q.mean(0)
338    _, _, vt = np.linalg.svd(c.astype(float), full_matrices=False)
339    off = c @ vt[1]  # signed distance from the fitted line
340    return {
341        "radius": float(np.median(radius[q[:, 0], q[:, 1]])),
342        "straightness": float(np.sqrt(np.mean(off**2))),
343        "tilt": float(np.degrees(np.arctan2(abs(vt[0][1]), abs(vt[0][0])))),
344    }

347def find_main_branches(tree: np.ndarray, skel: np.ndarray, params: GTParams = GTParams(),
348                       radius: np.ndarray | None = None) -> dict:
349    """Pick the root, fork and the two leaders from a tree skeleton.
350
351    Args:
352        tree: Tree region (used to locate the trunk base).
353        skel: Skeleton of the tree (connected).
354        params: Settings (``wander_penalty``, ``max_leader_radius``).
355        radius: Half-width image used for the post check; defaults to the
356            distance transform of ``tree``. Pass one computed from the full
357            foreground so leaders joined across gaps are measured too.
358
359    Returns:
360        Dict with ``root``, ``fork`` (``(row, col)`` or ``None``), ``trunk``,
361        ``left``, ``right`` paths (``(N, 2)`` arrays, possibly empty), their
362        ``trunk_shape``/``left_shape``/``right_shape`` and ``posts_skipped``.
363    """
364    empty = np.zeros((0, 2), int)
365    out = {"root": None, "fork": None, "trunk": empty, "left": empty, "right": empty,
366           "trunk_shape": {}, "left_shape": {}, "right_shape": {}, "posts_skipped": 0}
367    if skel.sum() < 2:
368        return out
369    ys, xs, a = _graph(skel)
370    ty, tx = np.nonzero(tree)
371    base_x = np.median(tx[ty >= np.percentile(ty, 97)])
372    # Root: skeleton point closest to the bottom centre of the trunk (sideways
373    # distance counts double, so stray ground pixels at the base are avoided).
374    root = int(np.argmin(np.hypot(ys - ty.max(), 2 * (xs - base_x))))
375    out["root"] = (int(ys[root]), int(xs[root]))
376    dist, pred = dijkstra(a, indices=root, return_predecessors=True)
377    ends = np.nonzero((np.diff(a.indptr) == 1) & np.isfinite(dist))[0]
378    ends = ends[ends != root]
379
380    def path(i: int) -> list[int]:
381        """Skeleton nodes from the root to node ``i``."""
382        p = [i]
383        while p[-1] != root:
384            p.append(int(pred[p[-1]]))
385        return p[::-1]
386
387    def score(e: int) -> float:
388        """A leader candidate's score: its rise minus a penalty for wandering."""
389        rise = ys[root] - ys[e]
390        return rise - params.wander_penalty * (dist[e] - rise)
391
392    coords = np.stack([ys, xs], 1)
393    if radius is None:
394        radius = ndi.distance_transform_edt(tree)
395
396    def best(candidates: list[int]) -> tuple[list[int] | None, dict]:
397        """Highest-scoring candidate that is not post-like."""
398        for e in sorted(candidates, key=score, reverse=True):
399            p = path(e)
400            shape = path_shape(coords[p], radius, params.shape_top_fraction)
401            if shape and shape["radius"] > params.max_leader_radius:
402                out["posts_skipped"] += 1
403                continue
404            return p, shape
405        return None, {}
406
407    pl, sl = best([e for e in ends if xs[e] < xs[root]])
408    pr, sr = best([e for e in ends if xs[e] >= xs[root]])
409    if pl is None or pr is None:
410        return out
411    shared = set(pr)
412    k = max(i for i, v in enumerate(pl) if v in shared)  # last shared node = fork
413    fork = pl[k]
414    kr = pr.index(fork)
415    out.update(
416        trunk_shape=path_shape(coords[pl[: k + 1]], radius, params.shape_top_fraction),
417        left_shape=sl,
418        right_shape=sr,
419        fork=(int(ys[fork]), int(xs[fork])),
420        trunk=coords[pl[: k + 1]],
421        left=coords[pl[k:]],
422        right=coords[pr[kr:]],
423    )
424    return out

427def extract_winter_gt(depth: np.ndarray, valid: np.ndarray, params: GTParams = GTParams()) -> WinterGT:
428    """Extract the trunk and the two V leaders from a winter depth image.
429
430    Args:
431        depth: ``(H, W)`` 8-bit depth (brighter = nearer).
432        valid: ``(H, W)`` bool validity mask.
433        params: Settings.
434
435    Returns:
436        A :class:`WinterGT`. ``ok`` is False (with reasons in ``issues``) if no
437        tree was found or a leader failed the quality check.
438    """
439    fg = foreground_mask(depth, valid, params)
440    tree = select_tree(fg, params)
441    if not tree.any():
442        return WinterGT(tree, np.zeros_like(tree), issues=["no tree region found"])
443    tree = ndi.binary_fill_holes(ndi.binary_closing(tree, disk(2)))
444
445    # Other pieces that gap linking may join.
446    others = fg & ~ndi.binary_dilation(tree, disk(1))
447    lab, n = ndi.label(others, structure=_K8)
448    areas = ndi.sum(others, lab, range(1, n + 1))
449    others = np.isin(lab, np.nonzero(areas >= params.min_piece_area)[0] + 1)
450
451    # Keep the largest connected piece of the tree's own skeleton.
452    sk = skeletonize(tree)
453    lab, n = ndi.label(sk, structure=_K8)
454    if n > 1:
455        sk = lab == (np.argmax(ndi.sum(sk, lab, range(1, n + 1))) + 1)
456    sk = link_gaps(sk, skeletonize(others), params)
457
458    radius = ndi.distance_transform_edt(ndi.binary_closing(fg, disk(2)) | tree)
459    br = find_main_branches(tree, sk, params, radius)
460    gt = WinterGT(tree, sk, **br)
461    if gt.fork is None:
462        gt.issues.append("could not find a (non-post) leader on each side")
463    else:
464        need = params.min_leader_rise * gt.root[0]
465        for side in ("left", "right"):
466            p = getattr(gt, side)
467            rise = p[0, 0] - p[-1, 0]
468            if rise < need:
469                gt.issues.append(f"{side} leader rises {rise} px (< {need:.0f})")
470    gt.ok = not gt.issues
471    return gt

474def render_main_branches(gt: WinterGT, min_radius: float = 2.0, max_radius: float = 25.0) -> np.ndarray:
475    """Draw the trunk and leaders as a mask with realistic widths.
476
477    If ``gt.radii`` is set (guided ground truth, fitted tapers), those radii
478    are used. Otherwise the radius at each path point is the distance from
479    that point to the edge of the tree region, clipped to
480    ``[min_radius, max_radius]``; linked gaps (outside the tree region) get
481    ``min_radius``. If ``gt.trunk_cuts`` is set, the trunk is cut flat
482    there; if ``gt.shoulders`` or ``gt.junction`` is set, those pixels are
483    added.
484
485    Args:
486        gt: Result of :func:`extract_winter_gt`.
487        min_radius: Smallest drawn radius in pixels.
488        max_radius: Largest drawn radius in pixels.
489
490    Returns:
491        Bool mask the same shape as ``gt.tree_mask``.
492    """
493    shape = gt.tree_mask.shape
494    out = np.zeros(shape, bool)
495    if gt.radii is None:
496        radius = np.clip(ndi.distance_transform_edt(gt.tree_mask), min_radius, max_radius)
497    for name in ("trunk", "left", "right"):
498        p = getattr(gt, name)
499        rs = gt.radii[name] if gt.radii is not None else radius[p[:, 0], p[:, 1]] if len(p) else []
500        part = np.zeros(shape, bool)
501        for (y, x), r in zip(p, rs):
502            rr, cc = draw_disk((y, x), r, shape=shape)
503            part[rr, cc] = True
504        if name == "trunk" and gt.trunk_cuts:
505            yy, xx = np.indices(shape)
506            for point, direction in gt.trunk_cuts:
507                part &= (yy - point[0]) * direction[0] + (xx - point[1]) * direction[1] <= 0
508        out |= part
509    for extra in (gt.shoulders, gt.junction):
510        if extra is not None:
511            out |= extra
512    return out

269              iters: int = 15) -> np.ndarray:
270    """Join skeleton pieces that continue a branch across a depth gap.
271
272    A branch end heading upwards is joined by a straight line to the nearest
273    pixel of another piece within ``link_gap`` and within ``link_max_angle``
274    of the end's direction; that whole piece is then added. Repeats until
275    nothing more joins.
276
277    Args:
278        main: Skeleton of the tree.
279        others: Skeletons of other foreground pieces.
280        params: Settings.
281        iters: Maximum number of passes.
282
283    Returns:
284        The extended skeleton.
285    """
286    lab, _ = ndi.label(others, structure=_K8)
287    oy, ox = np.nonzero(others)
288    skel = main.copy()
289    cos_min = np.cos(np.radians(params.link_max_angle))
290    for _ in range(iters):
291        added = False
292        for y, x, d in _end_directions(skel):
293            if -d[0] < params.link_min_up:  # row index decreases going up
294                continue
295            dy, dx = oy - y, ox - x
296            dist = np.hypot(dy, dx)
297            ok = (dist > 0) & (dist <= params.link_gap) & ~skel[oy, ox]
298            ok &= (dy * d[0] + dx * d[1]) >= cos_min * dist

311def path_shape(path: np.ndarray, radius: np.ndarray, top_fraction: float = 0.6) -> dict:
312    """Width, straightness and tilt of the upper part of a path.
313
314    The upper part is the points within the top ``top_fraction`` of the
315    path's rise. A straight line is fitted to it by total least squares (the
316    principal axis from an SVD of the centred points), which, unlike ordinary
317    regression, treats rows and columns alike and so also works for
318    near-vertical lines.
319
320    Args:
321        path: ``(N, 2)`` ``(row, col)`` points, starting at the bottom.
322        radius: Half-width image (distance from each pixel to the edge of the
323            tree region).
324        top_fraction: Fraction of the rise, from the top, to measure.
325
326    Returns:
327        ``{"radius": median half-width (px), "straightness": RMS distance from
328        the fitted line (px), "tilt": angle from vertical (degrees)}``, or an
329        empty dict if the path has fewer than 3 points.
330    """
331    p = np.asarray(path)
332    if len(p) < 3:
333        return {}
334    rise = p[0, 0] - p[:, 0].min()
335    q = p[p[:, 0] <= p[0, 0] - (1 - top_fraction) * rise]
336    if len(q) < 3:
337        q = p
338    c = q - q.mean(0)
339    _, _, vt = np.linalg.svd(c.astype(float), full_matrices=False)
340    off = c @ vt[1]  # signed distance from the fitted line
341    return {
342        "radius": float(np.median(radius[q[:, 0], q[:, 1]])),
343        "straightness": float(np.sqrt(np.mean(off**2))),
344        "tilt": float(np.degrees(np.arctan2(abs(vt[0][1]), abs(vt[0][0])))),
345    }

349                       radius: np.ndarray | None = None) -> dict:
350    """Pick the root, fork and the two leaders from a tree skeleton.
351
352    Args:
353        tree: Tree region (used to locate the trunk base).
354        skel: Skeleton of the tree (connected).
355        params: Settings (``wander_penalty``, ``max_leader_radius``).
356        radius: Half-width image used for the post check; defaults to the
357            distance transform of ``tree``. Pass one computed from the full
358            foreground so leaders joined across gaps are measured too.
359
360    Returns:
361        Dict with ``root``, ``fork`` (``(row, col)`` or ``None``), ``trunk``,
362        ``left``, ``right`` paths (``(N, 2)`` arrays, possibly empty), their
363        ``trunk_shape``/``left_shape``/``right_shape`` and ``posts_skipped``.
364    """
365    empty = np.zeros((0, 2), int)
366    out = {"root": None, "fork": None, "trunk": empty, "left": empty, "right": empty,
367           "trunk_shape": {}, "left_shape": {}, "right_shape": {}, "posts_skipped": 0}
368    if skel.sum() < 2:
369        return out
370    ys, xs, a = _graph(skel)
371    ty, tx = np.nonzero(tree)
372    base_x = np.median(tx[ty >= np.percentile(ty, 97)])
373    # Root: skeleton point closest to the bottom centre of the trunk (sideways
374    # distance counts double, so stray ground pixels at the base are avoided).
375    root = int(np.argmin(np.hypot(ys - ty.max(), 2 * (xs - base_x))))
376    out["root"] = (int(ys[root]), int(xs[root]))
377    dist, pred = dijkstra(a, indices=root, return_predecessors=True)
378    ends = np.nonzero((np.diff(a.indptr) == 1) & np.isfinite(dist))[0]
379    ends = ends[ends != root]
380
381    def path(i: int) -> list[int]:
382        """Skeleton nodes from the root to node ``i``."""
383        p = [i]
384        while p[-1] != root:
385            p.append(int(pred[p[-1]]))
386        return p[::-1]
387
388    def score(e: int) -> float:
389        """A leader candidate's score: its rise minus a penalty for wandering."""
390        rise = ys[root] - ys[e]
391        return rise - params.wander_penalty * (dist[e] - rise)
392
393    coords = np.stack([ys, xs], 1)
394    if radius is None:
395        radius = ndi.distance_transform_edt(tree)
396
397    def best(candidates: list[int]) -> tuple[list[int] | None, dict]:
398        """Highest-scoring candidate that is not post-like."""
399        for e in sorted(candidates, key=score, reverse=True):
400            p = path(e)
401            shape = path_shape(coords[p], radius, params.shape_top_fraction)
402            if shape and shape["radius"] > params.max_leader_radius:
403                out["posts_skipped"] += 1
404                continue
405            return p, shape
406        return None, {}
407
408    pl, sl = best([e for e in ends if xs[e] < xs[root]])
409    pr, sr = best([e for e in ends if xs[e] >= xs[root]])
410    if pl is None or pr is None:
411        return out
412    shared = set(pr)
413    k = max(i for i, v in enumerate(pl) if v in shared)  # last shared node = fork
414    fork = pl[k]
415    kr = pr.index(fork)
416    out.update(
417        trunk_shape=path_shape(coords[pl[: k + 1]], radius, params.shape_top_fraction),
418        left_shape=sl,
419        right_shape=sr,
420        fork=(int(ys[fork]), int(xs[fork])),
421        trunk=coords[pl[: k + 1]],
422        left=coords[pl[k:]],
423        right=coords[pr[kr:]],
424    )
425    return out

def extract_winter_gt( depth: numpy.ndarray, valid: numpy.ndarray, params: GTParams = GTParams(ground_row_fraction=0.5, ground_margin=6.0, min_tree_area=200, min_piece_area=30, link_gap=35.0, link_max_angle=35.0, link_min_up=0.5, wander_penalty=0.5, min_leader_rise=0.3, max_leader_radius=12.0, shape_top_fraction=0.6)) -> WinterGT: 
428def extract_winter_gt(depth: np.ndarray, valid: np.ndarray, params: GTParams = GTParams()) -> WinterGT:
429    """Extract the trunk and the two V leaders from a winter depth image.
430
431    Args:
432        depth: ``(H, W)`` 8-bit depth (brighter = nearer).
433        valid: ``(H, W)`` bool validity mask.
434        params: Settings.
435
436    Returns:
437        A :class:`WinterGT`. ``ok`` is False (with reasons in ``issues``) if no
438        tree was found or a leader failed the quality check.
439    """
440    fg = foreground_mask(depth, valid, params)
441    tree = select_tree(fg, params)
442    if not tree.any():
443        return WinterGT(tree, np.zeros_like(tree), issues=["no tree region found"])
444    tree = ndi.binary_fill_holes(ndi.binary_closing(tree, disk(2)))
445
446    # Other pieces that gap linking may join.
447    others = fg & ~ndi.binary_dilation(tree, disk(1))
448    lab, n = ndi.label(others, structure=_K8)
449    areas = ndi.sum(others, lab, range(1, n + 1))
450    others = np.isin(lab, np.nonzero(areas >= params.min_piece_area)[0] + 1)
451
452    # Keep the largest connected piece of the tree's own skeleton.
453    sk = skeletonize(tree)
454    lab, n = ndi.label(sk, structure=_K8)
455    if n > 1:
456        sk = lab == (np.argmax(ndi.sum(sk, lab, range(1, n + 1))) + 1)
457    sk = link_gaps(sk, skeletonize(others), params)
458
459    radius = ndi.distance_transform_edt(ndi.binary_closing(fg, disk(2)) | tree)
460    br = find_main_branches(tree, sk, params, radius)
461    gt = WinterGT(tree, sk, **br)
462    if gt.fork is None:
463        gt.issues.append("could not find a (non-post) leader on each side")
464    else:
465        need = params.min_leader_rise * gt.root[0]
466        for side in ("left", "right"):
467            p = getattr(gt, side)
468            rise = p[0, 0] - p[-1, 0]
469            if rise < need:
470                gt.issues.append(f"{side} leader rises {rise} px (< {need:.0f})")
471    gt.ok = not gt.issues
472    return gt
